cmake_minimum_required(VERSION 3.24)
project(DrivingBehaviorSimulator VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(SIM_SANITIZERS "Enable address/undefined sanitizers on GCC/Clang" OFF)
add_library(driving src/simulation/engine.cpp src/controllers/controllers.cpp src/scenarios/scenario.cpp src/metrics/metrics.cpp)
target_include_directories(driving PUBLIC include)
if(MSVC)
  target_compile_options(driving PUBLIC /W4 /permissive- /EHsc /utf-8)
else()
  target_compile_options(driving PUBLIC -Wall -Wextra -Wpedantic -Wconversion)
  if(SIM_SANITIZERS)
    target_compile_options(driving PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(driving PUBLIC -fsanitize=address,undefined)
  endif()
endif()
add_executable(simulator src/main.cpp)
target_link_libraries(simulator PRIVATE driving)
include(CTest)
if(BUILD_TESTING)
  add_executable(sim_tests tests/cpp/test_sim.cpp)
  target_link_libraries(sim_tests PRIVATE driving)
  add_test(NAME simulation_contracts COMMAND sim_tests)
endif()
